## AI-Friendly Manim Code Blueprint

This script is built for an AI or renderer to execute without 3D camera crashes or shader bugs:

```python
from manim import *

class CinematicTrialExplainer(ThreeDScene):
    def construct(self):
        # 1. Base Setup
        self.camera.background_color = "#0D0E15"
        
        # 2. Hero Card (Simple 2D Rounded Rectangle)
        card = RoundedRectangle(
            corner_radius=0.4, 
            height=4.2, 
            width=6.4, 
            color="#00F5D4", 
            fill_color="#131622", 
            fill_opacity=0.9,
            stroke_width=3
        )
        number_14 = Text("14", font="sans-serif", weight=BOLD, color="#FFFFFF").scale(2.2)
        
        # SCENE 1: Easy 3D Flip Entry (0:00 - 0:04)
        hero_group = VGroup(card, number_14)
        self.play(
            FadeIn(hero_group, scale=0.8),
            Rotate(hero_group, angle=PI, axis=UP),
            run_time=2.2,
            rate_func=smooth
        )
        self.wait(1.5)

        # SCENE 2 & 3: Compress to timeline, then re-expand (0:04 - 0:14)
        timeline = RoundedRectangle(corner_radius=0.15, height=0.3, width=6.4, color="#334155")
        dot = Dot(color="#00F5D4", radius=0.18).move_to(timeline.get_left())
        
        self.play(ReplacementTransform(card, timeline), FadeOut(number_14), run_time=1.5)
        self.play(dot.animate.move_to(timeline.get_right()), run_time=3.5, rate_func=smooth)
        
        # SCENE 6: Card returns and gets populated (0:27 - 0:34)
        reconstructed_card = RoundedRectangle(corner_radius=0.4, height=4.2, width=6.4, color="#00F5D4")
        b1 = RoundedRectangle(corner_radius=0.2, height=1.0, width=5.6, color="#00F5D4", fill_opacity=0.3).shift(UP * 1.1)
        b2 = RoundedRectangle(corner_radius=0.2, height=1.0, width=5.6, color="#FFB703", fill_opacity=0.3)
        b3 = RoundedRectangle(corner_radius=0.2, height=1.0, width=5.6, color="#7B2CBF", fill_opacity=0.3).shift(DOWN * 1.1)
        
        self.play(ReplacementTransform(timeline, reconstructed_card), FadeOut(dot), run_time=1.2)
        self.play(GrowFromCenter(b1), run_time=0.8)
        self.play(GrowFromCenter(b2), run_time=0.8)
        self.play(GrowFromCenter(b3), run_time=0.8)

        # SCENE 7: One Simple 2.5D Camera Tilt (0:34 - 0:38)
        self.move_camera(phi=40 * DEGREES, run_time=3, rate_func=smooth)
        self.wait(1)

        # SCENE 8: The "Keep" Confirmation (0:39 - 0:44)
        keep_btn = RoundedRectangle(corner_radius=0.2, height=0.6, width=2.4, color="#FFFFFF", fill_color="#00F5D4", fill_opacity=1).shift(DOWN * 1.1)
        btn_text = Text("KEEP", font="sans-serif", weight=BOLD, color="#000000").scale(0.5).move_to(keep_btn)
        
        self.play(FadeIn(keep_btn), Write(btn_text), run_time=1.2)
        self.play(keep_btn.animate.scale(1.1), run_time=0.3)
        self.play(keep_btn.animate.scale(1/1.1), run_time=0.3)
        self.wait(2)
